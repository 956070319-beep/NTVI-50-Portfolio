    private void OnAutoRunCameraFrameArrived(BitmapSource bitmap)
    {
        if (bitmap is null)
            return;

        Interlocked.Exchange(ref _pendingCameraFrame, bitmap);
        ScheduleAutoRunCameraFrame();
    }

    private void ScheduleAutoRunCameraFrame()
    {
        if (Interlocked.Exchange(ref _cameraFrameDispatchPending, 1) != 0)
            return;

        try
        {
            Dispatcher.BeginInvoke(DispatcherPriority.Background, new Action(ApplyLatestAutoRunCameraFrame));
        }
        catch (InvalidOperationException)
        {
            Interlocked.Exchange(ref _cameraFrameDispatchPending, 0);
        }
    }

    private void ApplyLatestAutoRunCameraFrame()
    {
        try
        {
            var bitmap = Interlocked.Exchange(ref _pendingCameraFrame, null);
            if (bitmap is null || !IsLoaded)
                return;


            ImgAutoCamera.Source = bitmap;
            TxtAutoImagePlaceholder.Visibility = Visibility.Collapsed;
            TxtAutoImageInfo.Text = $"\u5b9e\u65f6\u56fe\u50cf\uff1a{DateTime.Now:HH:mm:ss.fff}";
        }
        finally
        {
            Interlocked.Exchange(ref _cameraFrameDispatchPending, 0);
            if (IsLoaded && Volatile.Read(ref _pendingCameraFrame) is not null)
                ScheduleAutoRunCameraFrame();
        }
    }
